import { SecretVault } from '@secretvault/sdk';

// Zero secrets saved to disk or .env files
const vault = await SecretVault.connect({
  appId: 'payments-api',
  auth: {
    method: 'oidc-workload',
    issuer: 'https://vault.acme.corp'
  },
  cache: 'ephemeral-mlock', // prevented from swapping to disk
  autoRotate: true
});

// Hydrate strictly into private memory segment
const dbUrl = await vault.getSecret('DATABASE_URL');
const stripeKey = await vault.getSecret('STRIPE_SECRET_KEY');
