package secretvault.invariants.storage

# Deny persistent disk mounts for non-tmpfs confidential envelopes
default allow := false

allow {
    input.target_environment == "production"
    input.mount_type == "tmpfs"
    input.encryption_standard == "AES-256-GCM-RAM"
    not disk_persistence_detected
}

disk_persistence_detected {
    input.storage_drivers[_].persisted == true
}

violation[msg] {
    disk_persistence_detected
    msg := sprintf(
        "Hard Invariant Violation: Plaintext or block storage detected for %v in %v",
        [input.secret_id, input.cluster_id]
    )
}
