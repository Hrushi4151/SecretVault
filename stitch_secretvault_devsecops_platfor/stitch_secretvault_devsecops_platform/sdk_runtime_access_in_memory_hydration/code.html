import { SecretVault } from '@secretvault/sdk';

// Zero-knowledge in-memory hydration
const vault = new SecretVault({
  enclaveAttestation: true,
  cacheTTL: '15m',
  onRotation: (secret, newVersion) => {
    console.log(`[SecretVault] Hot-reloaded ${secret.name} to version ${newVersion}`);
  }
});

const secrets = await vault.load({
  environment: 'production',
  project: 'payments-api',
  fallbackToCached: true
});
